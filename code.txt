  if (typeof value === 'string' && /^\d+$/.test(value)) value = Number(value);
  return Number.isInteger(value) && value > 0 && value <= 2_147_483_647 ? value : null;
}

<file path="middlewares/authRateLimit.js">
const WINDOW_MS = 10 * 60 * 1000;
const MAX_IP_KEYS = 10_000;
const MAX_ACCOUNT_KEYS = 20_000;
const byIp = new Map();
const byAccount = new Map();
let resetAt = Date.now() + WINDOW_MS;

function allowAttempt(attempts, key, limit, maxKeys) {
  const count = attempts.get(key) ?? 0;
  if (count >= limit || (!count && attempts.size >= maxKeys)) return false;
  attempts.set(key, count + 1);
  return true;
}

function limitCredentialAttempts(req, res, next) {
  const now = Date.now();
  if (now >= resetAt) {
    byIp.clear();
    byAccount.clear();
    resetAt = now + WINDOW_MS;
  }

  const address = req.ip ?? req.socket.remoteAddress ?? 'unknown';
  const username = req.body?.username;
  let accountKey;
  if (req.user) accountKey = `${address}\0id:${req.user.userId}`;
  else if (typeof username === 'string' && username.length <= 1024) {
    accountKey = `${address}\0name:${username.trim().toLowerCase()}`;
  }
  if (!allowAttempt(byIp, address, 240, MAX_IP_KEYS) ||
      (accountKey && !allowAttempt(byAccount, accountKey, 15, MAX_ACCOUNT_KEYS))) {
    res.set('Retry-After', String(Math.ceil((resetAt - now) / 1000)));
    return res.status(429).json({ error: 'Too many requests. Try again later.' });
  }
  next();
}

module.exports = { limitCredentialAttempts };

const express = require('express');
const { authenticateToken, readAuthenticatedUser } = require('../middlewares/auth');
const {
  getSnapshot, registerDevice, heartbeat, acknowledge, applyCommand, subscribe, PlaybackError,
} = require('../services/playback');

const router = express.Router();
const owners = new Map();
const MAX_STREAMS = 64;
const MAX_OWNER_STREAMS = 8;
const REFRESH_INTERVAL = 2_500;
const KEEPALIVE_INTERVAL = 12_000;
const STREAM_DURATION = 60_000;
let openStreams = 0;

router.use(authenticateToken, (req, res, next) => {
  res.set('Cache-Control', 'no-store');
  next();
});

function playbackEndpoint(handler) {
  return async (req, res, next) => {
    const { userId } = req.user;
    try {
      const snapshot = await handler(userId, req.body);
      res.json(snapshot);
    } catch (error) {
      if (!(error instanceof PlaybackError)) return next(error);
      if (error.status !== 409) return res.status(error.status).json({ error: error.message });
      try {
        const snapshot = await getSnapshot(userId);
        return res.status(409).json({ error: error.message, snapshot });
      } catch (readError) {
        return next(readError);
      }
    }
  };
}

function eventVersion(snapshot) {
  const { session, devices } = snapshot;
  return JSON.stringify([
    session.version, session.leaseUntil, session.playing, session.positionAt,
    devices.map(device => [device.id, device.name, device.platform, device.lastSeenAt, device.online]),
  ]);
}

router.get('/session', playbackEndpoint(getSnapshot));
router.post('/devices', playbackEndpoint(registerDevice));
router.post('/heartbeat', playbackEndpoint(heartbeat));
router.post('/ack', playbackEndpoint(acknowledge));
router.post('/command', playbackEndpoint(applyCommand));

router.get('/events', (req, res) => {
  const { userId } = req.user;
  const count = owners.get(userId) ?? 0;
  if (openStreams >= MAX_STREAMS || count >= MAX_OWNER_STREAMS) {
    return res.set('Retry-After', '5').status(429).json({ error: 'Too many open playback streams' });
  }
  owners.set(userId, count + 1);
  openStreams++;
  res.set({
    'Content-Type': 'text/event-stream; charset=utf-8',
    'Cache-Control': 'private, no-store, no-transform',
    'X-Accel-Buffering': 'no',
  });
  let closed = false;
  let refreshing = false;
  let needsRefresh = false;
  let lastEvent;

  function finish() {
    if (closed) return;
    closed = true;
    clearInterval(refreshTimer);
    clearInterval(keepaliveTimer);
    clearTimeout(expiryTimer);
    unsubscribe();
    openStreams--;
    const remaining = owners.get(userId) - 1;
    if (remaining) owners.set(userId, remaining);
    else owners.delete(userId);
    if (!res.destroyed && !res.writableEnded) res.end();
  }

  function send(value) {
    if (closed) return;
    if (res.destroyed || !res.write(value)) finish();
  }

  async function refresh() {
    if (closed) return;
    if (refreshing) {
      needsRefresh = true;
      return;
    }
    refreshing = true;
    try {
      do {
        needsRefresh = false;
        const snapshot = await getSnapshot(userId);
        if (closed) return;
        const version = eventVersion(snapshot);
        if (version !== lastEvent) {
          const auth = await readAuthenticatedUser(req.headers.authorization);
          if (closed) return;
          if (auth?.userId !== userId) return finish();
          lastEvent = version;
          send(`event: session\ndata: ${JSON.stringify(snapshot)}\n\n`);
        }
      } while (needsRefresh && !closed);
    } catch (error) {
      console.warn('Playback event stream refresh failed:', error.code || error.name);
      finish();
    } finally {
      refreshing = false;
    }
  }

  const unsubscribe = subscribe(userId, refresh);
  const refreshTimer = setInterval(refresh, REFRESH_INTERVAL);
  const keepaliveTimer = setInterval(() => send(': keepalive\n\n'), KEEPALIVE_INTERVAL);
  const expiryTimer = setTimeout(finish, STREAM_DURATION);
  res.once('close', finish);
  if (res.destroyed) return finish();
  res.flushHeaders();
  send('retry: 2000\n\n');
  void refresh();
});

const PNG_HEADER = Buffer.from([0x89, 0x50, 0x4e, 0x47, 0x0d, 0x0a, 0x1a, 0x0a]);
const WEBP_CHUNKS = new Set(['VP8 ', 'VP8L', 'VP8X']);

function detectImageType(buffer) {
  if (!Buffer.isBuffer(buffer)) return null;
  if (buffer.length >= 4 && buffer[0] === 0xff && buffer[1] === 0xd8 && buffer[2] === 0xff) {
    return { contentType: 'image/jpeg', extension: '.jpg' };
  }
  if (buffer.length >= 24 && buffer.subarray(0, 8).equals(PNG_HEADER) && buffer.toString('ascii', 12, 16) === 'IHDR') {
    return { contentType: 'image/png', extension: '.png' };
  }
  const gifHeader = buffer.toString('ascii', 0, 6);
  if (buffer.length >= 13 && (gifHeader === 'GIF87a' || gifHeader === 'GIF89a')) {
    return { contentType: 'image/gif', extension: '.gif' };
  }
  if (buffer.length >= 16 && buffer.toString('ascii', 0, 4) === 'RIFF' &&
      buffer.toString('ascii', 8, 12) === 'WEBP' && WEBP_CHUNKS.has(buffer.toString('ascii', 12, 16))) {
    return { contentType: 'image/webp', extension: '.webp' };
  }
  return null;
}

const { createHash, timingSafeEqual } = require('node:crypto');
const { EventEmitter } = require('node:events');
const prisma = require('../lib/prisma');
const {
  ONLINE_MS, PlaybackError, emptySession, hasLiveLease, leaseDeadline, boundedPosition,
  projectedPosition, presentSession, optionalPosition, validatePlaybackCommand, validateDeviceId,
  validatePositionReport, shuffledOrder, navigationCandidates,
} = require('./playbackState');

const MAX_DEVICES = 20;
const DEVICE_RETENTION_MS = 30 * 24 * 60 * 60 * 1000;
const platforms = new Set(['browser', 'telegram']);
const events = new EventEmitter();

function subscribe(userId, listener) {
  const key = String(userId);
  events.on(key, listener);
  return () => events.off(key, listener);
}

function publish(userId) {
  events.emit(String(userId));
}

async function databaseTime(connection = prisma) {
  const [row] = await connection.$queryRaw`SELECT clock_timestamp() AS "now"`;
  return row.now;
}

function deviceWhere(userId, id) {
  return { userId_id: { userId, id } };
}

function deviceCredentials(body) {
  const id = validateDeviceId(body?.deviceId);
  const { deviceKey } = body;
  if (typeof deviceKey !== 'string' || !/^[a-z\d_-]{32,128}$/i.test(deviceKey)) {
    throw new PlaybackError(400, 'Invalid device key');
  }
  const keyHash = createHash('sha256').update(deviceKey).digest('hex');
  return { id, keyHash };
}

function matchesDevice(device, credentials) {
  return Boolean(device && device.keyHash.length === credentials.keyHash.length &&
    timingSafeEqual(Buffer.from(device.keyHash), Buffer.from(credentials.keyHash)));
}

function expectedVersion(body) {
  const version = body?.expectedVersion;
  if (!Number.isInteger(version) || version < 0 || version > 2147483647) {
    throw new PlaybackError(400, 'Invalid session version');
  }
  return version;
}

function isOnline(device, now) {
  return Boolean(device && now.getTime() - device.lastSeenAt.getTime() < ONLINE_MS);
}

async function getSnapshot(userId) {
  const [session, devices] = await Promise.all([
    prisma.playbackSession.findUnique({ where: { userId } }),
    prisma.playbackDevice.findMany({
      where: { userId },
      select: { id: true, name: true, platform: true, lastSeenAt: true },
      orderBy: [{ lastSeenAt: 'desc' }, { id: 'asc' }],
      take: MAX_DEVICES,
    }),
  ]);
  const now = await databaseTime();
  return {
    serverTime: now.getTime(),
    session: presentSession(session ?? emptySession, now),
    devices: devices.map(device => ({
      ...device,
      lastSeenAt: device.lastSeenAt.getTime(),
      online: isOnline(device, now),
    })),
  };
}

async function registerDevice(userId, body) {
  const credentials = deviceCredentials(body);
  const name = typeof body.name === 'string' ? body.name.trim() : '';
  if (!name || name.length > 80 || /[\x00-\x1f\x7f]/.test(name)) {
    throw new PlaybackError(400, 'Invalid device name');
  }
  if (!platforms.has(body.platform)) throw new PlaybackError(400, 'Invalid device platform');

  await prisma.$transaction(async tx => {
    // Serialize registration and eviction for this owner to enforce the roster cap.
    const owners = await tx.$queryRaw`SELECT "id" FROM "User" WHERE "id" = ${userId} FOR UPDATE`;
    if (!owners.length) throw new PlaybackError(401, 'Account no longer exists');
    const now = await databaseTime(tx);
    const where = deviceWhere(userId, credentials.id);
    const device = await tx.playbackDevice.findUnique({ where });
    const data = { name, platform: body.platform, lastSeenAt: now };
    if (device) {
      if (!matchesDevice(device, credentials)) throw new PlaybackError(403, 'Device credentials do not match');
      await tx.playbackDevice.update({ where, data });
      return;
    }

    const session = await tx.playbackSession.upsert({ where: { userId }, create: { userId }, update: {} });
    const protectedIds = [session.activeDeviceId, session.pendingDeviceId].filter(Boolean);
    const eligible = { userId, id: { notIn: protectedIds } };
    await tx.playbackDevice.deleteMany({
      where: { ...eligible, lastSeenAt: { lt: new Date(now.getTime() - DEVICE_RETENTION_MS) } },
    });
    if (await tx.playbackDevice.count({ where: { userId } }) >= MAX_DEVICES) {
      const offline = { ...eligible, lastSeenAt: { lt: new Date(now.getTime() - ONLINE_MS) } };
      const oldest = await tx.playbackDevice.findFirst({ where: offline, orderBy: { lastSeenAt: 'asc' } });
      if (!oldest) throw new PlaybackError(409, 'Too many playback devices are online');
      const removed = await tx.playbackDevice.deleteMany({ where: { ...offline, id: oldest.id } });
      if (!removed.count) throw new PlaybackError(409, 'Device roster changed');
    }
    await tx.playbackDevice.create({ data: { ...data, ...credentials, userId } });
  });
  publish(userId);
  return getSnapshot(userId);
}

async function verifyDevice(userId, body, touch = false) {
  const credentials = deviceCredentials(body);
  const device = await prisma.playbackDevice.findUnique({ where: deviceWhere(userId, credentials.id) });
  if (!matchesDevice(device, credentials)) throw new PlaybackError(403, 'Unknown playback device');
  if (touch) {
    const now = await databaseTime();
    if (now > device.lastSeenAt) {
      const touched = await prisma.playbackDevice.updateMany({
        where: { userId, id: device.id, keyHash: credentials.keyHash, lastSeenAt: { lt: now } },
        data: { lastSeenAt: now },
      });
      if (!touched.count) {
        const current = await prisma.playbackDevice.findUnique({ where: deviceWhere(userId, credentials.id) });
        if (!matchesDevice(current, credentials)) throw new PlaybackError(403, 'Unknown playback device');
      }
    }
  }
  return device.id;
}

async function mutateSession(userId, version, transition) {
  for (let attempt = 0; attempt < 5; attempt++) {
    const session = await prisma.playbackSession.findUnique({ where: { userId } });
    if (!session) throw new PlaybackError(403, 'No registered playback session');
    if (version !== undefined && session.version !== version) {
      throw new PlaybackError(409, 'Playback session changed');
    }
    const now = await databaseTime();
    const mutation = await transition(session, now);
    if (!mutation) return;
    const { changes, extendOnly } = mutation;
    const result = await prisma.playbackSession.updateMany({
      // Heartbeats leave version stable. The lease belongs in the CAS so a handoff
      // cannot shorten an already-issued permit for the outgoing audio device.
      where: {
        userId, version: session.version, leaseUntil: session.leaseUntil,
        position: session.position, positionAt: session.positionAt,
      },
      data: extendOnly ? changes : { ...changes, version: { increment: 1 } },
    });
    if (result.count) {
      publish(userId);
      return;
    }
  }
  throw new PlaybackError(409, 'Playback session changed');
}

function claim(session, now, deviceId) {
  if (hasLiveLease(session, now)) return {};
  return { activeDeviceId: deviceId, pendingDeviceId: null, handoffResume: false, leaseUntil: leaseDeadline(now) };
}

function finishHandoff(session, now, position) {
  return {
    activeDeviceId: session.pendingDeviceId,
    pendingDeviceId: null,
    leaseUntil: leaseDeadline(now),
    handoffResume: false,
    playing: session.handoffResume && Boolean(session.trackId),
    position: boundedPosition(position ?? projectedPosition(session, now), session.duration),
    positionAt: now,
  };
}

async function heartbeat(userId, body) {
  const deviceId = await verifyDevice(userId, body, true);
  const report = validatePositionReport(body.report);
  await mutateSession(userId, undefined, (session, now) => {
    const live = hasLiveLease(session, now);
    if (!live && session.pendingDeviceId === deviceId) {
      return { changes: finishHandoff(session, now) };
    }
    if (live && session.activeDeviceId === deviceId && !session.pendingDeviceId) {
      const changes = { leaseUntil: leaseDeadline(now) };
      if (session.playing && report?.version === session.version && report.trackId === session.trackId) {
        changes.position = boundedPosition(report.position, session.duration);
        changes.positionAt = now;
      }
      return { changes, extendOnly: true };
    }
    return null;
  });
  return getSnapshot(userId);
}

async function acknowledge(userId, body) {
  const deviceId = await verifyDevice(userId, body);
  const version = expectedVersion(body);
  const position = optionalPosition(body.position);
  await mutateSession(userId, version, (session, now) => {
    if (session.activeDeviceId !== deviceId) throw new PlaybackError(403, 'Only the outgoing device may acknowledge');
    if (!session.pendingDeviceId) throw new PlaybackError(409, 'No pending handoff');
    return { changes: finishHandoff(session, now, position) };
  });
  return getSnapshot(userId);
}

async function catalogTracks(ids) {
  if (!ids.length) return new Map();
  const tracks = await prisma.track.findMany({
    where: { id: { in: ids } },
    select: { id: true, duration: true },
  });
  return new Map(tracks.map(track => [track.id, track]));
}

async function playChanges(session, now, deviceId, command) {
  let queue = command.queue ?? session.queue;
  const trackId = command.trackId ?? (command.queue ? queue[0] : session.trackId);
  if (!trackId) throw new PlaybackError(409, 'Select a track first');
  if (!queue.includes(trackId)) queue = [trackId];
  const tracks = await catalogTracks(queue);
  if (!tracks.has(trackId)) throw new PlaybackError(404, 'Track no longer exists');
  queue = queue.filter(id => tracks.has(id));
  const index = queue.indexOf(trackId);
  const duration = tracks.get(trackId).duration;
  const sameTrack = session.trackId === trackId;
  const unchangedQueue = queue.length === session.queue.length && queue.every((id, i) => id === session.queue[i]);
  const position = command.position ?? (sameTrack ? projectedPosition(session, now) : 0);
  const playingFrom = duration !== null && position >= duration ? 0 : position;
  return {
    ...claim(session, now, deviceId),
    trackId,
    duration,
    queue,
    queueIndex: index,
    shuffleOrder: session.shuffle && !(sameTrack && unchangedQueue)
      ? shuffledOrder(queue, index)
      : session.shuffleOrder,
    playing: true,
    position: boundedPosition(playingFrom, duration),
    positionAt: now,
  };
}

async function navigationChanges(session, now, deviceId, command) {
  const fromEnded = command.type === 'ended';
  const position = projectedPosition(session, now);
  const live = hasLiveLease(session, now);
  if (fromEnded && (session.activeDeviceId !== deviceId || !live)) {
    throw new PlaybackError(403, 'Only the active device may finish a track');
  }
  if (fromEnded && command.trackId !== undefined && command.trackId !== session.trackId) {
    throw new PlaybackError(409, 'Track already changed');
  }
  if (!session.trackId || (fromEnded && !session.playing)) return null;
  if (fromEnded && session.repeat === 'one') return { playing: true, position: 0, positionAt: now };
  if (command.type === 'previous' && position > 3) {
    return { playing: live && session.playing, position: 0, positionAt: now };
  }

  const candidates = navigationCandidates(session, command.type, fromEnded);
  const tracks = await catalogTracks(candidates.map(index => session.queue[index]));
  const index = candidates.find(candidate => tracks.has(session.queue[candidate]));
  if (index === undefined) {
    return { playing: false, position: fromEnded ? session.duration ?? position : position, positionAt: now };
  }
  const trackId = session.queue[index];
  return {
    ...claim(session, now, deviceId), trackId, duration: tracks.get(trackId).duration,
    queueIndex: index, playing: true, position: 0, positionAt: now,
  };
}

async function transferChanges(userId, session, now, targetDeviceId) {
  const live = hasLiveLease(session, now);
  if (live && targetDeviceId === session.pendingDeviceId) return null;
  if (live && session.pendingDeviceId) throw new PlaybackError(409, 'Handoff is already pending');
  if (live && targetDeviceId === session.activeDeviceId) return null;

  const target = await prisma.playbackDevice.findUnique({ where: deviceWhere(userId, targetDeviceId) });
  if (!target) throw new PlaybackError(403, 'Unknown handoff target');
  if (!isOnline(target, now)) throw new PlaybackError(409, 'Handoff device is offline');
  if (live) return { pendingDeviceId: targetDeviceId, handoffResume: session.playing };

  return finishHandoff({
    ...session,
    pendingDeviceId: targetDeviceId,
    handoffResume: session.pendingDeviceId ? session.handoffResume : false,
  }, now);
}

async function commandChanges(userId, session, now, deviceId, command) {
  if (command.type === 'transfer') return transferChanges(userId, session, now, command.targetDeviceId);

  const live = hasLiveLease(session, now);
  if (session.pendingDeviceId && (live || command.type !== 'play')) {
    throw new PlaybackError(409, 'Handoff is pending');
  }
  if (session.pendingDeviceId && deviceId !== session.pendingDeviceId) {
    const target = await prisma.playbackDevice.findUnique({ where: deviceWhere(userId, session.pendingDeviceId) });
    if (isOnline(target, now)) throw new PlaybackError(409, 'Handoff is pending');
  }

  switch (command.type) {
    case 'play':
      return playChanges(session, now, deviceId, command);
    case 'pause':
    case 'release': {
      if (command.type === 'release' && session.activeDeviceId !== deviceId) {
        throw new PlaybackError(403, 'Only the active device may release playback');
      }
      const reported = session.activeDeviceId === deviceId ? command.position : undefined;
      const position = boundedPosition(reported ?? projectedPosition(session, now), session.duration);
      const released = command.type === 'release' ? { activeDeviceId: null, leaseUntil: null } : {};
      return { ...released, playing: false, position, positionAt: now };
    }
    case 'seek':
      if (!session.trackId) throw new PlaybackError(409, 'Select a track first');
      return { playing: live && session.playing, position: boundedPosition(command.position, session.duration), positionAt: now };
    case 'next':
    case 'previous':
    case 'ended':
      return navigationChanges(session, now, deviceId, command);
    case 'setShuffle':
      if (session.shuffle === command.shuffle) return null;
      return { shuffle: command.shuffle, shuffleOrder: command.shuffle ? shuffledOrder(session.queue, session.queueIndex) : [] };
    case 'setRepeat':
      return session.repeat === command.repeat ? null : { repeat: command.repeat };
  }
}

async function applyCommand(userId, body) {
  const deviceId = await verifyDevice(userId, body, true);
  const version = expectedVersion(body);
  const command = validatePlaybackCommand(body.command);
  await mutateSession(userId, version, async (session, now) => {
    const changes = await commandChanges(userId, session, now, deviceId, command);
    return changes ? { changes } : null;
  });
  return getSnapshot(userId);
}

module.exports = { getSnapshot, registerDevice, heartbeat, acknowledge, applyCommand, subscribe, PlaybackError };

const { positiveInt } = require('../lib/validation');

const LEASE_MS = 12_000;
const ONLINE_MS = 30_000;
const MAX_QUEUE = 300;
const MAX_POSITION = 7 * 24 * 60 * 60;
const repeatOptions = new Set(['none', 'all', 'one']);

class PlaybackError extends Error {
  constructor(status, message) {
    super(message);
    this.status = status;
  }
}

const emptySession = {
  version: 0, activeDeviceId: null, pendingDeviceId: null, leaseUntil: null,
  handoffResume: false, trackId: null, duration: null, queue: [], queueIndex: -1,
  shuffleOrder: [], playing: false, position: 0, positionAt: new Date(0), shuffle: false,
  repeat: 'none',
};

function hasLiveLease(session, now) {
  return Boolean(session.activeDeviceId && session.leaseUntil && session.leaseUntil > now);
}

function leaseDeadline(now) {
  return new Date(now.getTime() + LEASE_MS);
}

function boundedPosition(position, duration) {
  return Math.min(Math.max(position, 0), duration ?? MAX_POSITION);
}

function projectedPosition(session, now) {
  if (!session.trackId) return 0;
  const limit = session.leaseUntil ? Math.min(now.getTime(), session.leaseUntil.getTime()) : 0;
  const elapsed = session.playing ? Math.max(0, limit - session.positionAt.getTime()) / 1000 : 0;
  return boundedPosition(session.position + elapsed, session.duration);
}

function presentSession(session, now) {
  return {
    version: session.version,
    activeDeviceId: session.activeDeviceId,
    pendingDeviceId: session.pendingDeviceId,
    leaseUntil: session.leaseUntil?.getTime() ?? null,
    trackId: session.trackId,
    duration: session.duration,
    queue: session.queue,
    queueIndex: session.queueIndex,
    playing: !session.pendingDeviceId && hasLiveLease(session, now) && session.playing,
    position: projectedPosition(session, now),
    positionAt: session.positionAt.getTime(),
    shuffle: session.shuffle,
    repeat: session.repeat,
  };
}

function optionalPosition(position) {
  if (position === undefined) return undefined;
  if (typeof position !== 'number' || !Number.isFinite(position) || position < 0 || position > MAX_POSITION) {
    throw new PlaybackError(400, 'Invalid playback position');
  }
  return position;
}

function validateTrackId(value) {
  const id = positiveInt(value);
  if (!id) throw new PlaybackError(400, 'Invalid track ID');
  return id;
}

function validatePositionReport(report) {
  if (report === undefined) return null;
  if (!report || !Number.isInteger(report.version) || report.version < 0 || report.version > 2147483647) {
    throw new PlaybackError(400, 'Invalid position report');
  }
  const position = optionalPosition(report.position);
  if (position === undefined) throw new PlaybackError(400, 'Playback position required');
  return { trackId: validateTrackId(report.trackId), version: report.version, position };
}

function validatePlaybackCommand(command) {
  if (!command || typeof command !== 'object') throw new PlaybackError(400, 'Invalid playback command');
  const { type } = command;
  switch (type) {
    case 'play': {
      const trackId = command.trackId === undefined ? undefined : validateTrackId(command.trackId);
      let queue;
      if (command.queue !== undefined) {
        if (!Array.isArray(command.queue) || !command.queue.length || command.queue.length > MAX_QUEUE) {
          throw new PlaybackError(400, 'Invalid queue size');
        }
        queue = [...new Set(command.queue.map(validateTrackId))];
        if (trackId && !queue.includes(trackId)) throw new PlaybackError(400, 'Track is not in the queue');
      }
      return { type, trackId, queue, position: optionalPosition(command.position) };
    }
    case 'pause':
    case 'release':
      return { type, position: optionalPosition(command.position) };
    case 'seek': {
      const position = optionalPosition(command.position);
      if (position === undefined) throw new PlaybackError(400, 'Playback position required');
      return { type, position };
    }
    case 'next':
    case 'previous':
      return { type };
    case 'ended':
      return { type, trackId: command.trackId === undefined ? undefined : validateTrackId(command.trackId) };
    case 'setShuffle':
      if (typeof command.shuffle !== 'boolean') throw new PlaybackError(400, 'Invalid shuffle setting');
      return { type, shuffle: command.shuffle };
    case 'setRepeat':
      if (!repeatOptions.has(command.repeat)) throw new PlaybackError(400, 'Invalid repeat setting');
      return { type, repeat: command.repeat };
    case 'transfer':
      return { type, targetDeviceId: validateDeviceId(command.targetDeviceId) };
    default:
      throw new PlaybackError(400, 'Unknown playback command');
  }
}

function validateDeviceId(value) {
  if (typeof value !== 'string' || !/^[a-f\d]{8}-[a-f\d]{4}-[a-f\d]{4}-[a-f\d]{4}-[a-f\d]{12}$/i.test(value)) {
    throw new PlaybackError(400, 'Invalid device ID');
  }
  return value.toLowerCase();
}

function shuffledOrder(queue, first) {
  if (!queue.length) return [];
  const remaining = queue.map((trackId, index) => index).filter(index => index !== first);
  for (let end = remaining.length - 1; end > 0; end--) {
    const swap = Math.floor(Math.random() * (end + 1));
    [remaining[end], remaining[swap]] = [remaining[swap], remaining[end]];
  }
  return first < 0 ? remaining : [first, ...remaining];
}

function navigationCandidates(session, direction, fromEnded) {
  const order = session.shuffle && session.shuffleOrder.length === session.queue.length
    ? session.shuffleOrder : session.queue.map((trackId, index) => index);
  const position = order.indexOf(session.queueIndex);
  if (direction === 'previous') return order.slice(0, Math.max(position, 0)).reverse().concat(order.slice(0, 1));
  const following = order.slice(position + 1);
  if (fromEnded && session.repeat !== 'all') return following;
  return following.concat(order.slice(0, position + 1));
}

};

    await tx.telegramLinkCode.upsert({

      create: { userId, codeHash, expiresAt },
      update: { codeHash, expiresAt },
    });
  });
  return { code, expiresAt };
}

async function linkAccounts(db, codeHash, telegramId, now) {
  const link = await db.telegramLinkCode.findUnique({
    where: { codeHash },
    select: { user: { select: linkableFields }, expiresAt: true },
  });
  if (!link || link.expiresAt <= now) throw expiredLink();
  const target = link.user;
  if (!target.username || !target.password || target.telegramId) throw expiredLink();
  const duplicate = await db.user.findUnique({ where: { telegramId }, select: linkableFields });
  if (duplicate && (duplicate.isAdmin || duplicate.password || target.isAdmin)) {
    throw new TelegramLinkError(409, 'This Telegram account cannot be merged');
  }
  return { target, duplicate };
}

async function previewTelegramLink(code, telegramId) {
  const { target, duplicate } = await linkAccounts(prisma, hashCode(code), telegramId, new Date());
  return { username: target.username, mergeRequired: Boolean(duplicate) };
}

async function moveAccountData(tx, from, to) {
  const favorites = await tx.favorite.findMany({
    where: { userId: from },
    select: { trackId: true, createdAt: true, updatedAt: true },
  });
  if (favorites.length) {
    await tx.favorite.createMany({ data: favorites.map((favorite) => ({ ...favorite, userId: to })), skipDuplicates: true });
    await tx.favorite.deleteMany({ where: { userId: from } });
  }
  await tx.review.updateMany({ where: { userId: from }, data: { userId: to } });
  await tx.complaint.updateMany({ where: { userId: from }, data: { userId: to } });
}

async function confirmTelegramLink(code, identity, confirmMerge) {
  const codeHash = hashCode(code);
  const result = await serializable(async (tx) => {
    const now = new Date();
    const { target, duplicate } = await linkAccounts(tx, codeHash, identity.telegramId, now);
    if (duplicate && !confirmMerge) throw new TelegramLinkError(409, 'Confirm the account merge');
    const consumed = await tx.telegramLinkCode.deleteMany({
      where: { userId: target.id, codeHash, expiresAt: { gt: now } },
    });
    if (consumed.count !== 1) throw expiredLink();

    const profileImageUrl = target.profileImageUrl ?? duplicate?.profileImageUrl ?? null;
    if (duplicate) {
      await moveAccountData(tx, duplicate.id, target.id);
      await tx.user.delete({ where: { id: duplicate.id } });
    }
    const user = await tx.user.update({
      where: { id: target.id },
      data: { ...identity, profileImageUrl, telegramAuthVersion: { increment: 1 } },
      select: { id: true, isAdmin: true, telegramId: true, telegramAuthVersion: true },
    });
    return { user, unusedAvatar: duplicate?.profileImageUrl !== profileImageUrl ? duplicate : null };
  });
  if (result.unusedAvatar) await cleanupUserAvatar(result.unusedAvatar.id, result.unusedAvatar.profileImageUrl);
  return result.user;
}

async function unlinkTelegram(userId, password) {
  const verified = await passwordAccount(userId, password);
  if (!verified.telegramId) throw new TelegramLinkError(409, 'Telegram is not linked');
  return serializable(async (tx) => {
    const user = await tx.user.update({
      where: {
        id: userId,
        password: verified.password,
        telegramId: verified.telegramId,
        telegramAuthVersion: verified.telegramAuthVersion,
      },
      data: {
        telegramId: null, telegramName: null, telegramUsername: null, telegramPhotoUrl: null,
        telegramAuthVersion: { increment: 1 },
      },
      select: { id: true, isAdmin: true },
    });
    await tx.telegramLinkCode.deleteMany({ where: { userId } });
    return user;
  });
}

const { randomUUID } = require('node:crypto');
const { createReadStream } = require('node:fs');
const path = require('node:path');
const prisma = require('../lib/prisma');
const { positiveInt } = require('../lib/validation');
const { detectImageType } = require('./images');

const AUDIO_LIMIT = 100 * 1024 * 1024;
const IMAGE_LIMIT = 5 * 1024 * 1024;
const BATCH_LIMIT = 200 * 1024 * 1024;
const MAX_TRACKS = 5;
const TEXT_LIMIT = 300;

class TrackError extends Error {
  constructor(status, message) {
    super(message);
    this.status = status;
  }
}

function trackId(value) {
  const id = positiveInt(value);
  if (id === null) throw new TrackError(400, 'Invalid track ID');
  return id;
}

function trackText(value) {
  return typeof value === 'string' ? value.trim().slice(0, TEXT_LIMIT) : '';
}

function normalizeDuration(value) {
  return Number.isFinite(value) && value > 0 && value <= 2147483647 ? Math.round(value) : null;
}

function audioDetails(format) {
  if (format.hasVideo || !format.sampleRate || !format.numberOfChannels) return null;
  const { container = '', codec = '' } = format;
  if (/^MPEG.*Layer (?:I{1,3}|[123])\b/i.test(codec)) return { extension: '.mp3', contentType: 'audio/mpeg' };
  if (container === 'FLAC') return { extension: '.flac', contentType: 'audio/flac' };
  if (container === 'Ogg' && /Vorbis|Opus|Speex|FLAC/i.test(codec)) return { extension: '.ogg', contentType: 'audio/ogg' };
  if (container === 'WAVE') return { extension: '.wav', contentType: 'audio/wav' };
  if (/^ADTS\b/i.test(container)) return { extension: '.aac', contentType: 'audio/aac' };
  if (/(?:AAC|ALAC)/i.test(codec)) return { extension: '.m4a', contentType: 'audio/mp4' };
  return null;
}

function inspectImage(buffer, required = false) {
  if (Buffer.isBuffer(buffer) && buffer.length > IMAGE_LIMIT && required) {
    throw new TrackError(413, 'Image exceeds 5 MiB');
  }
  const details = Buffer.isBuffer(buffer) && buffer.length <= IMAGE_LIMIT && detectImageType(buffer);
  if (details) return { buffer, ...details };
  if (required) throw new TrackError(400, 'Unsupported image file');
  return null;
}

async function prepareTrack({ audio, image, overrides = {}, fallback = {}, loadFallbackCover }) {
  if (!audio?.size) throw new TrackError(400, 'Audio file required');
  if (audio.size > AUDIO_LIMIT) throw new TrackError(413, 'Audio file exceeds 100 MiB');

  let info;
  try {
    const metadata = await import('music-metadata');
    info = audio.path
      ? await metadata.parseFile(audio.path, { duration: true })
      : await metadata.parseBuffer(audio.buffer, undefined, { duration: true });
  } catch {
    throw new TrackError(400, 'Invalid or unsupported audio file');
  }

  const details = audioDetails(info.format);
  if (!details) throw new TrackError(400, 'Unsupported audio format');
  const source = audio.originalname || '';
  const name = path.parse(path.win32.basename(source)).name;
  const tags = info.common;
  const cover = image
    ? inspectImage(image, true)
    : tags.picture?.map(item => inspectImage(Buffer.from(item.data.buffer, item.data.byteOffset, item.data.byteLength))).find(Boolean);
  const fallbackCover = !cover && loadFallbackCover ? await loadFallbackCover() : null;

  return {
    audio: { ...audio, ...details },
    cover: cover || inspectImage(fallbackCover),
    data: {
      title: trackText(overrides.title) || trackText(tags.title) || trackText(fallback.title) || trackText(name) || 'Unknown',
      artist: trackText(overrides.artist) || trackText(tags.artist) || trackText(fallback.artist) || 'Unknown',
      album: trackText(overrides.album) || trackText(tags.album) || null,
      genre: trackText(overrides.genre) || trackText(tags.genre?.[0]) || null,
      duration: normalizeDuration(info.format.duration) ?? normalizeDuration(fallback.duration),
    },
  };
}

function bucket(name) {
  return require('../lib/supabase').storage.from(name);
}

function publicStorageKey(value, bucketName) {
  if (typeof value !== 'string' || !process.env.SUPABASE_URL) return null;
  try {
    const source = new URL(value);
    const base = new URL(process.env.SUPABASE_URL);
    const prefix = `${base.pathname.replace(/\/$/, '')}/storage/v1/object/public/${bucketName}/`;
    if (source.origin !== base.origin || source.username || source.password || !source.pathname.startsWith(prefix)) return null;
    const parts = source.pathname.slice(prefix.length).split('/').map(decodeURIComponent);
    if (parts.some(part => !part || part === '.' || part === '..' || /[\\/\x00-\x1f]/.test(part))) return null;
    if (bucketName === 'images' && parts[0] === 'avatars') return null;
    return parts.join('/');
  } catch {
    return null;
  }
}

async function removeObjects(objects) {
  for (const { name, key } of objects) {
    try {
      const { error } = await bucket(name).remove([key]);
      if (error) throw error;
    } catch (error) {
      console.warn('Track storage cleanup failed', { bucket: name, key, message: error.message });
    }
  }
}

async function uploadObject(name, key, file, attempted) {
  const storage = bucket(name);
  attempted.push({ name, key });
  const stream = file.path ? createReadStream(file.path) : null;
  try {
    const { error } = await storage.upload(key, stream || file.buffer, {
      contentType: file.contentType,
      upsert: false,
    });
    if (error) throw error;
    return storage.getPublicUrl(key).data.publicUrl;
  } finally {
    stream?.destroy();
  }
}

async function saveTracks(inputs) {
  if (!inputs.length || inputs.length > MAX_TRACKS) throw new TrackError(400, 'Invalid track count');
  if (inputs.reduce((total, input) => total + (input.audio?.size || 0), 0) > BATCH_LIMIT) {
    throw new TrackError(413, 'Total audio size exceeds 200 MiB');
  }

  const attempted = [];
  try {
    const records = [];
    for (const input of inputs) {
      const { audio, cover, data } = await prepareTrack(input);
      const name = randomUUID();
      const filename = await uploadObject('tracks', name + audio.extension, audio, attempted);
      const imageUrl = cover ? await uploadObject('images', name + cover.extension, cover, attempted) : null;
      records.push({ ...data, filename, imageUrl });
    }
    if (records.length === 1) return [await prisma.track.create({ data: records[0] })];
    return await prisma.$transaction(records.map(data => prisma.track.create({ data })));
  } catch (error) {
    await removeObjects(attempted);
    throw error;
  }
}

function legacyName(value, directory) {
  if (typeof value !== 'string') return null;
  const prefix = new RegExp(`^/?(?:public/)?${directory}/`);
  const name = value.replace(/\\/g, '/').replace(prefix, '');
  if (!name || name.startsWith('.') || /[\\/:*?"<>|\x00-\x1f]/.test(name) || /[. ]$/.test(name)) return null;
  return name;
}

function coverUrl(value) {
  if (typeof value === 'string' && /^https?:\/\//i.test(value)) return value;
  const name = legacyName(value, 'images');
  return name ? `/images/${encodeURIComponent(name)}` : null;
}

async function updateTrack(value, fields) {
  const id = trackId(value);
  const data = {};
  for (const field of ['title', 'artist', 'album', 'genre']) {
    const value = fields[field];
    if (value === undefined) continue;
    const nullable = field === 'album' || field === 'genre';
    if (value === null && nullable) {
      data[field] = null;
      continue;
    }
    if (typeof value !== 'string' || value.trim().length > TEXT_LIMIT || (!nullable && !value.trim())) {
      throw new TrackError(400, `Invalid ${field}`);
    }
    data[field] = value.trim() || null;
  }
  try {
    return await prisma.track.update({ where: { id }, data });
  } catch (error) {
    if (error.code === 'P2025') throw new TrackError(404, 'Track not found');
    throw error;
  }
}

async function deleteTrack(value) {
  const id = trackId(value);
  let track;
  try {
    track = await prisma.$transaction(async tx => {
      await tx.favorite.deleteMany({ where: { trackId: id } });
      await tx.review.deleteMany({ where: { trackId: id } });
      return tx.track.delete({ where: { id } });
    });
  } catch (error) {
    if (error.code === 'P2025') throw new TrackError(404, 'Track not found');
    throw error;
  }
  const audio = publicStorageKey(track.filename, 'tracks');
  const cover = publicStorageKey(track.imageUrl, 'images');
  const objects = [];
  if (audio) objects.push({ name: 'tracks', key: audio });
  if (cover) objects.push({ name: 'images', key: cover });
  await removeObjects(objects);
}

  AUDIO_LIMIT, BATCH_LIMIT, IMAGE_LIMIT, MAX_TRACKS,
  TrackError, trackId, publicStorageKey, legacyName, coverUrl, saveTracks, updateTrack, deleteTrack,
};

function signedTelegramLaunch(id) {
  const authDate = String(Math.floor(Date.now() / 1_000));

        }
      } else {

        buffer += decoder.decode(value, { stream: true });
      }
    }
  };

const { createHmac, timingSafeEqual } = require('node:crypto');

const MAX_AGE_SECONDS = 24 * 60 * 60;
const CLOCK_SKEW_SECONDS = 60;
const MAX_INIT_DATA_LENGTH = 16_384;

// https://core.telegram.org/bots/webapps#validating-data-received-via-the-mini-app
function readTelegramUser(initData, botToken, now = Math.floor(Date.now() / 1000), maxAgeSeconds = MAX_AGE_SECONDS) {
  if (typeof initData !== 'string' || initData.length === 0 || initData.length > MAX_INIT_DATA_LENGTH) {
    return null;
  }
  const params = new URLSearchParams(initData);
  const fields = new Map();
  for (const [key, value] of params) {
    if (fields.has(key) || !/^[A-Za-z0-9_]+$/.test(key) || /[\r\n]/.test(value)) return null;
    fields.set(key, value);
  }

  const hash = fields.get('hash');
  const authDate = fields.get('auth_date');
  if (!hash || !/^[a-f0-9]{64}$/i.test(hash) || !authDate || !/^\d+$/.test(authDate)) return null;
  const timestamp = Number(authDate);
  if (!Number.isSafeInteger(timestamp) || timestamp <= 0 || now - timestamp > maxAgeSeconds ||
      timestamp - now > CLOCK_SKEW_SECONDS) return null;

  fields.delete('hash');
  const checkString = [...fields.keys()].sort().map((key) => `${key}=${fields.get(key)}`).join('\n');
  const secret = createHmac('sha256', 'WebAppData').update(botToken).digest();
  const expected = createHmac('sha256', secret).update(checkString).digest();
  if (!timingSafeEqual(expected, Buffer.from(hash, 'hex'))) return null;

  try {
    const user = JSON.parse(fields.get('user'));
    if (!user || !Number.isSafeInteger(user.id) || user.id <= 0 || typeof user.first_name !== 'string') {
      return null;
    }
    if (['last_name', 'username', 'photo_url'].some((field) => user[field] != null && typeof user[field] !== 'string')) {
      return null;
    }
    return user;
  } catch {
    return null;
  }
}

module.exports = { readTelegramUser };

const jwt = require('jsonwebtoken');

function issueToken(user) {
  const claims = { userId: user.id, isAdmin: user.isAdmin };
  if (user.telegramId) {
    claims.telegramId = user.telegramId;
    claims.telegramAuthVersion = user.telegramAuthVersion ?? 0;
  }
  return jwt.sign(claims, process.env.JWT_SECRET, { algorithm: 'HS256', expiresIn: '12h' });
}

function verifyToken(token) {
  const claims = jwt.verify(token, process.env.JWT_SECRET, { algorithms: ['HS256'] });
  if (!Number.isInteger(claims?.userId) || claims.userId <= 0 || claims.userId > 2147483647 ||
      !Number.isFinite(claims.exp)) {
    throw new Error('Invalid token claims');
  }
  if (claims.telegramId !== undefined && (typeof claims.telegramId !== 'string' || !/^\d+$/.test(claims.telegramId))) {
    throw new Error('Invalid Telegram token claims');
  }
  return claims;
}

function managedAvatar(bucket, avatarUrl) {
  if (!avatarUrl) return null;
  try {
    const root = new URL(bucket.getPublicUrl('avatars/').data.publicUrl);
    const url = new URL(avatarUrl);
    if (url.origin !== root.origin || !url.pathname.startsWith(root.pathname)) return null;
    const filename = url.pathname.slice(root.pathname.length);
    const match = /^([1-9]\d*)_(?:\d+|[a-f0-9-]{36})\.(?:jpg|png|webp|gif)$/.exec(filename);
    const owner = Number(match?.[1]);
    return Number.isInteger(owner) && owner <= 2147483647 ? { owner, path: `avatars/${filename}` } : null;
  } catch {
    return null;
  }
}

  const bucket = imageBucket();
  const avatar = managedAvatar(bucket, profileImageUrl);
  if (!avatar) return;
  if (avatar.owner !== id) {
    // Merged profiles may retain an avatar from a deleted Telegram-only account.
    try {
      const existing = await prisma.user.findFirst({
        where: { OR: [{ id: avatar.owner }, { profileImageUrl }] },
        select: { id: true },
      });
      if (existing) return;
    } catch {
      console.warn('Transferred avatar ownership could not be confirmed');
      return;
    }
  }
  await removeImage(bucket, avatar.path);
}

async function deleteUserAccount(id) {
  let user;
  try {
    user = await prisma.$transaction(async (tx) => {
      await tx.favorite.deleteMany({ where: { userId: id } });
      await tx.review.deleteMany({ where: { userId: id } });
      await tx.complaint.deleteMany({ where: { userId: id } });
      return tx.user.delete({ where: { id }, select: { profileImageUrl: true } });
    });
  } catch (error) {
    if (error.code === 'P2025') return false;
    throw error;
  }
  await cleanupUserAvatar(id, user.profileImageUrl);
  return true;
}

async function updateUserAvatar(id, buffer, { extension, contentType }) {
  const where = { id };
  const select = { profileImageUrl: true };
  const current = await prisma.user.findUnique({ where, select });
  if (!current) return { status: 'missing' };

  const bucket = imageBucket();
  const path = `avatars/${id}_${randomUUID()}${extension}`;
  const profileImageUrl = bucket.getPublicUrl(path).data.publicUrl;
  const { error } = await bucket.upload(path, buffer, { contentType, upsert: false });
  if (error) throw new Error('Avatar storage upload failed');

  let updated;
  try {
    const { count } = await prisma.user.updateMany({
      where: { id, profileImageUrl: current.profileImageUrl },
      data: { profileImageUrl },
    });
    updated = count === 1;
  } catch (updateError) {
    // A database timeout can arrive after the change committed. Confirm before deleting the new file.
    const saved = await prisma.user.findUnique({ where, select }).catch(() => undefined);
    if (saved === undefined) throw updateError;
    if (saved?.profileImageUrl !== profileImageUrl) {
      await removeImage(bucket, path);
      throw updateError;
    }
    updated = true;
  }

  if (!updated) {
    await removeImage(bucket, path);
    return { status: 'conflict' };
  }

  await cleanupUserAvatar(id, current.profileImageUrl);
  return { status: 'saved', profileImageUrl };
}

# music-streaming-server

Express API for the website and Telegram Mini App. PostgreSQL stores users, tracks and favorites. Supabase Storage serves uploaded audio, cover art and avatars.

## Run locally

Use Node.js 22 or newer. Copy `.env.example` to `.env` and fill in the database URLs, a private JWT signing secret and Supabase server credentials. Use a development database for local migration runs.

```powershell
npm ci
npx prisma generate
npx prisma migrate deploy
npm start
```

- `DATABASE_URL` is the application connection. `DIRECT_URL` is the direct PostgreSQL connection used for migrations.
- `SUPABASE_URL` and `SUPABASE_SERVICE_KEY` are required on the backend. Existing `tracks` and `images` buckets need public reads for playback and cover art. Storage writes use the server key.
- `TELEGRAM_BOT_TOKEN` enables Telegram login and bot polling. Music upload is available in private chat to Telegram-linked administrators or Telegram IDs in the comma-separated `TELEGRAM_UPLOAD_USER_IDS` list.
- `CORS_ORIGINS` adds exact frontend origins as a comma-separated list. The production UNDERLAB origin and the default local Vite origins are already allowed.
- Set `TRUST_PROXY_HOPS` only when the number of trusted reverse-proxy hops is known. The credential attempt limiter uses the client IP resolved by Express.

## Catalog and audio

Apply the Prisma migration before enabling Realtime in the frontend. On Supabase it publishes the public track catalog and keeps account, favorite, complaint and review tables behind the API. It also adds four foreign-key indexes.

`GET /tracks/:id/stream` serves bundled legacy tracks with HTTP Range and redirects new tracks to Supabase Storage. The bot and administrator upload routes share file validation and metadata extraction.

Run `npm test` for the HTTP API checks. They start an ephemeral loopback server with database fixtures.

## Shared playback

To connect an existing site login to Telegram, create a one-time code from the profile on the site and confirm it in the Mini App. The code expires after five minutes. Linking can merge a separate Telegram-only account with explicit consent. The website account keeps its ID. Favorites are deduplicated and reviews and complaints are moved. Unlinking requires the website password.

The `/playback` API stores a versioned session per user and treats each open page as a playback device. The authenticated event stream sends updates to that user's devices. Devices can control the queue from any connected page. One device holds the audio lease at a time. Handoff pauses the outgoing device before granting the new lease and falls back to expiry when the outgoing device loses its connection.

const prisma = require('../lib/prisma');
const { verifyToken } = require('../services/tokens');

function unauthorized(res) {
  return res.set('WWW-Authenticate', 'Bearer').status(401).json({ error: 'Authentication required' });
}

async function readAuthenticatedUser(authorizationHeader) {
  const match = /^Bearer\s+(\S+)$/i.exec(authorizationHeader ?? '');
  if (!match) return null;
  let claims;
  try {
    claims = verifyToken(match[1]);
  } catch {
    return null;
  }
  if (claims.telegramId) {
    const current = await prisma.user.findUnique({
      where: { id: claims.userId },
      select: { telegramId: true, telegramAuthVersion: true },
    });
    if (!current || current.telegramId !== claims.telegramId ||
        (current.telegramAuthVersion ?? 0) !== (claims.telegramAuthVersion ?? 0)) {
      return null;
    }
  }
  return { userId: claims.userId };
}

async function authenticateToken(req, res, next) {
  try {
    const user = await readAuthenticatedUser(req.headers.authorization);
    if (!user) return unauthorized(res);
    req.user = user;
  } catch (error) {
    return next(error);
  }
  next();
}

async function requireAdmin(req, res, next) {
  if (!req.user) return unauthorized(res);
  try {
    const user = await prisma.user.findUnique({
      where: { id: req.user.userId },
      select: { isAdmin: true },
    });
    if (!user) return unauthorized(res);
    if (!user.isAdmin) return res.status(403).json({ error: 'Forbidden' });
    req.user.isAdmin = true;
  } catch (error) {
    return next(error);
  }
  next();
}

module.exports = { authenticateToken, readAuthenticatedUser, requireAdmin };

        displayName: user.telegramName,
        profileImageUrl: user.profileImageUrl ?? user.telegramPhotoUrl ?? null,
        telegramId: user.telegramId,
      },

function linkIdentity(initData) {
  const botToken = process.env.TELEGRAM_BOT_TOKEN;
  if (!botToken) throw new TelegramLinkError(503, 'Telegram login unavailable');
  const user = readTelegramUser(initData, botToken, undefined, LINK_INIT_DATA_MAX_AGE);
  if (!user) throw new TelegramLinkError(401, 'Reopen the Mini App to confirm Telegram again');
  return telegramIdentity(user);
}

const { mkdtemp, rm } = require('node:fs/promises');
const { tmpdir } = require('node:os');
const path = require('node:path');
const multer = require('multer');
const { AUDIO_LIMIT, BATCH_LIMIT, IMAGE_LIMIT, MAX_TRACKS } = require('../services/tracks');

const uploadRoot = path.resolve(tmpdir());
const maxActiveUploads = 2;
const maxRequestSize = BATCH_LIMIT + IMAGE_LIMIT + 1024 * 1024;
let activeUploads = 0;

exports.upload = multer({
  storage: multer.diskStorage({
    destination: (req, file, callback) => callback(null, req.trackUploadDirectory),
  }),
  limits: {
    fileSize: AUDIO_LIMIT,
    files: MAX_TRACKS,
    fields: 4,
    fieldSize: 1024,
    parts: MAX_TRACKS + 5,
  },
});

function parseUpload(parser, req, res) {
  return new Promise((resolve, reject) => {
    const onAbort = () => reject(new Error('Track upload was aborted'));
    req.once('aborted', onAbort);
    parser(req, res, error => {
      req.removeListener('aborted', onAbort);
      if (error) reject(error);
      else resolve();
    });
  });
}

exports.withTrackUpload = (parser, handler) => async (req, res, next) => {
  const requestSize = Number(req.get('content-length'));
  if (requestSize > maxRequestSize) return res.status(413).json({ error: 'Upload exceeds the configured limit' });
  if (activeUploads >= maxActiveUploads) {
    return res.set('Retry-After', '5').status(503).json({ error: 'Track upload is busy. Try again shortly.' });
  }

  let directory;
  activeUploads++;
  try {
    directory = await mkdtemp(path.join(uploadRoot, 'underlab-tracks-'));
    if (req.aborted) return;
    req.trackUploadDirectory = directory;
    await parseUpload(parser, req, res);
    if (!req.aborted) await handler(req, res);
  } catch (error) {
    if (!req.aborted) next(error);
  } finally {
    if (directory && path.dirname(directory) === uploadRoot) {
      try {
        await rm(directory, { recursive: true, force: true, maxRetries: 3 });
      } catch (error) {
        console.warn('Temporary track cleanup failed:', error.code);
      }
    }
    activeUploads--;
  }
};

exports.uploadAvatar = multer({
  storage: multer.memoryStorage(),
  limits: { fileSize: IMAGE_LIMIT, files: 1, fields: 0 },
  fileFilter: (req, file, callback) => {
    if (!file.mimetype.startsWith('image/')) {
      return callback(new multer.MulterError('LIMIT_UNEXPECTED_FILE', file.fieldname));
    }
    callback(null, true);
  },
});

const { Bot } = require('node-telegram-bot-api');
const prisma = require('../lib/prisma');
const { IMAGE_LIMIT, TrackError, saveTracks } = require('../services/tracks');

const TELEGRAM_AUDIO_LIMIT = 20 * 1024 * 1024;

async function downloadFile(api, token, file, limit) {
  if (file.file_size > limit) throw new TrackError(413, 'This file exceeds the Telegram download limit.');
  const info = await api.getFile({ file_id: file.file_id });
  if (info.file_size > limit) throw new TrackError(413, 'This file exceeds the Telegram download limit.');
  const parts = info.file_path?.split('/');
  if (!parts?.length || parts.some(part => !part || part === '.' || part === '..')) {
    throw new Error('Telegram did not return a valid file path');
  }
  const filePath = parts.map(encodeURIComponent).join('/');
  const response = await fetch(`https://api.telegram.org/file/bot${token}/${filePath}`, {
    redirect: 'error',
    signal: AbortSignal.timeout(60_000),
  });
  if (!response.ok || !response.body || Number(response.headers.get('content-length')) > limit) {
    await response.body?.cancel();
    throw new TrackError(400, 'Telegram could not provide a downloadable file within the limit.');
  }
  const chunks = [];
  let size = 0;
  for await (const chunk of response.body) {
    size += chunk.byteLength;
    if (size > limit) throw new TrackError(413, 'This file exceeds the Telegram download limit.');
    chunks.push(chunk);
  }
  return Buffer.concat(chunks, size);
}

function launch() {
  const token = process.env.TELEGRAM_BOT_TOKEN;
  if (!token) {
    console.log('Telegram bot disabled');
    return;
  }
  const uploaders = new Set((process.env.TELEGRAM_UPLOAD_USER_IDS || '').split(',')
    .map(value => value.trim()).filter(value => /^\d+$/.test(value)));
  const bot = new Bot(token);
  let busy = false;

  bot.command('start', ctx => ctx.reply(
    'Send an audio file in this private chat to add it to the music library. Upload access is required.\n\nMP3, FLAC, AAC, OGG, WAV, M4A',
  ));

  bot.on('message', async ctx => {
    const message = ctx.message;
    const media = message?.audio || message?.document;
    if (!media || !ctx.from?.id || ctx.chat?.type !== 'private') return;
    const telegramId = String(ctx.from.id);
    if (!uploaders.has(telegramId)) {
      const user = await prisma.user.findUnique({ where: { telegramId }, select: { isAdmin: true } });
      if (!user?.isAdmin) {
        await ctx.reply('This account cannot add tracks to the library.');
        return;
      }
    }
    if (busy) {
      await ctx.reply('Another track is being uploaded. Try again shortly.');
      return;
    }

    busy = true;
    let result;
    try {
      const buffer = await downloadFile(ctx.api, token, media, TELEGRAM_AUDIO_LIMIT);
      const thumbnail = media.thumbnail || media.thumb;
      const [track] = await saveTracks([{
        audio: { buffer, size: buffer.length, originalname: media.file_name || '' },
        fallback: { title: media.title, artist: media.performer, duration: media.duration },
        loadFallbackCover: thumbnail ? () => downloadFile(ctx.api, token, thumbnail, IMAGE_LIMIT).catch(() => null) : undefined,
      }]);
      const details = [track.title, track.artist, track.album, track.genre].filter(Boolean).join('\n');
      result = `Track added\n\n${details}\n\nID: ${track.id}`;
    } catch (error) {
      console.error('Telegram track upload failed:', error.code || error.name);
      result = error instanceof TrackError ? error.message : 'Could not save the track. Try again later.';
    } finally {
      busy = false;
    }
    await ctx.reply(result);
  });

  bot.catch(error => console.error('Telegram update failed:', error.name));
  void bot.init().then(() => {
    console.log('Telegram bot started');
    return bot.startPolling(undefined, { allowedUpdates: ['message'] });
  }).catch(error => console.error('Telegram polling failed:', error.name));
  return bot;
}

module.exports = { launch };

exports.addFavorite = async (req, res) => {
  const trackId = positiveInt(req.body?.trackId);
  if (!trackId) return res.status(400).json({ error: 'Invalid track ID' });

  const userId = req.user.userId;
  try {
    const favorite = await prisma.favorite.create({ data: { trackId, userId } });
    return res.status(201).json({ message: 'Трек добавлен в избранное', favorite });
  } catch (error) {
    if (error.code === 'P2002') {
      const favorite = await prisma.favorite.findUnique({ where: { userId_trackId: { userId, trackId } } });
      return res.json({ message: 'Трек уже добавлен в избранное', favorite });
    }
    if (error.code === 'P2003') return res.status(404).json({ error: 'Track or user not found' });
    throw error;
  }
};

exports.removeFavorite = async (req, res) => {
  const trackId = positiveInt(req.body?.trackId);
  if (!trackId) return res.status(400).json({ error: 'Invalid track ID' });

  await prisma.favorite.deleteMany({ where: { userId: req.user.userId, trackId } });
  return res.json({ message: 'Трек удален из избранного' });
};

exports.getFavorites = async (req, res) => {
  const favorites = await prisma.favorite.findMany({
    where: { userId: req.user.userId },
    select: {
      favoriteId: true,
      trackId: true,
      track: {
        select: {
          title: true, artist: true, album: true, genre: true, duration: true,
          imageUrl: true, filename: true, createdAt: true,
        },
      },
    },
    orderBy: { favoriteId: 'desc' },
  });
  return res.json({
    favorites: favorites.map(({ track, ...favorite }) => ({ ...favorite, ...track })),
  });
};

const path = require('node:path');

const {
  IMAGE_LIMIT, TrackError, trackId, publicStorageKey, legacyName, coverUrl,
  saveTracks, updateTrack, deleteTrack,
} = require('../services/tracks');

const tracksDirectory = path.join(__dirname, '..', 'public', 'tracks');
const legacyAudioExtensions = new Set(['.mp3', '.mp2', '.flac', '.m4a', '.mp4', '.aac', '.ogg', '.oga', '.opus', '.wav', '.aiff']);

function serializeTrack(track) {
  return { ...track, imageUrl: coverUrl(track.imageUrl) };
}

function sendError(res, error, message) {
  if (error instanceof TrackError) return res.status(error.status).json({ error: error.message });
  console.error(message, error.code || error.name);
  return res.status(500).json({ error: message });
}

exports.addTrack = async (req, res) => {
  try {
    const cover = req.files?.image?.[0];
    const audio = req.files?.track?.[0];
    if (!cover || !audio) throw new TrackError(400, 'Both image and audio file are required');
    if (cover.size > IMAGE_LIMIT) throw new TrackError(413, 'Image exceeds 5 MiB');
    const image = await fs.readFile(cover.path);
    const [track] = await saveTracks([{ audio, image, overrides: req.body }]);
    res.status(201).json({ message: 'Track uploaded', track: serializeTrack(track) });
  } catch (error) {
    sendError(res, error, 'Failed to add track');
  }
};

exports.getTracks = async (req, res) => {
  try {
    const tracks = await prisma.track.findMany();
    for (let i = tracks.length - 1; i > 0; i--) {
      const j = Math.floor(Math.random() * (i + 1));
      [tracks[i], tracks[j]] = [tracks[j], tracks[i]];
    }
    res.json(tracks.map(serializeTrack));
  } catch (error) {
    sendError(res, error, 'Failed to get tracks');
  }
};

exports.streamTrack = async (req, res, next) => {
  try {
    const id = trackId(req.params.id);
    const track = await prisma.track.findUnique({ where: { id } });
    if (!track) throw new TrackError(404, 'Track not found');
    res.set('X-Content-Type-Options', 'nosniff');
    if (publicStorageKey(track.filename, 'tracks')) {
      return res.redirect(302, track.filename);
    }

    const name = legacyName(track.filename, 'tracks');
    if (!name || !legacyAudioExtensions.has(path.extname(name).toLowerCase())) {
      throw new TrackError(404, 'Audio file not found');
    }
    const file = await fs.lstat(path.join(tracksDirectory, name)).catch(error => {
      if (error.code === 'ENOENT') return null;
      throw error;
    });
    if (!file?.isFile()) throw new TrackError(404, 'Audio file not found');
    return res.sendFile(name, { root: tracksDirectory, dotfiles: 'deny' }, error => {
      if (!error || res.destroyed || error.code === 'ECONNABORTED') return;
      if (res.headersSent) return next(error);
      if (error.status === 416) {
        res.set('Content-Range', `bytes */${file.size}`);
        return res.status(416).end();
      }
      const status = error.status === 404 ? 404 : 500;
      return res.status(status).json({ error: status === 404 ? 'Audio file not found' : 'Failed to stream track' });
    });
  } catch (error) {
    sendError(res, error, 'Failed to stream track');
  }
};

exports.search = async (req, res) => {
  const query = typeof req.query.query === 'string' ? req.query.query.trim() : '';
  if (!query) return res.status(400).json({ error: 'Query parameter required' });
  if (query.length > 300) return res.status(400).json({ error: 'Query is too long' });
  const searchTerm = query.replace(/[\\%_]/g, '\\$&');
  try {
    const tracks = await prisma.track.findMany({
      where: {
        OR: ['title', 'artist', 'album'].map(field => ({
          [field]: { contains: searchTerm, mode: 'insensitive' },
        })),
      },
    });
    res.json(tracks.map(serializeTrack));
  } catch (error) {
    sendError(res, error, 'Search failed');
  }
};

exports.uploadMultipleTracks = async (req, res) => {
  try {
    if (!req.files?.length) throw new TrackError(400, 'No files provided');
    const tracks = await saveTracks(req.files.map(audio => ({ audio })));
    res.status(201).json({ message: 'Tracks uploaded', tracks: tracks.map(serializeTrack) });
  } catch (error) {
    sendError(res, error, 'Bulk upload failed');
  }
};

exports.editTrack = async (req, res) => {
  try {
    const track = await updateTrack(req.params.id, req.body || {});
    res.status(200).json({ message: 'Track updated', track: serializeTrack(track) });
  } catch (error) {
    sendError(res, error, 'Failed to update track');

};

exports.deleteTrack = async (req, res) => {
  try {
    await deleteTrack(req.params.id);
    res.status(200).json({ message: 'Track deleted' });
  } catch (error) {
    sendError(res, error, 'Failed to delete track');
  }
};

const bcrypt = require('bcrypt');
const prisma = require('../lib/prisma');
const { positiveInt } = require('../lib/validation');

const { issueToken } = require('../services/tokens');

const SALT_ROUNDS = 10;
const DUMMY_HASH = '$2b$10$aRm61gwI7yrSl2dQhC9DAOwynHuw5p1eq0OrtxCr4ezoLFTGz5SpC';
const publicUserFields = {
  id: true,
  username: true,
  telegramName: true,
  profileImageUrl: true,
  createdAt: true,
};

function newUsername(value) {
  if (typeof value !== 'string') return null;
  const username = value.trim();
  return username.length >= 1 && username.length <= 64 && !/[\0-\x1f\x7f]/.test(username) ? username : null;
}

function isNewPassword(password) {
  return typeof password === 'string' && password.length >= 8 && Buffer.byteLength(password) <= 72 &&
    !password.includes('\0');
}

function isExistingPassword(password) {
  return typeof password === 'string' && password.length > 0 && Buffer.byteLength(password) <= 1024;
}

exports.register = async (req, res, next) => {
  const { username, password } = req.body ?? {};
  const normalizedUsername = newUsername(username);
  if (!normalizedUsername) return res.status(400).json({ error: 'Username must be 1–64 characters' });
  if (!isNewPassword(password)) {
    return res.status(400).json({ error: 'Password must be at least 8 characters and at most 72 bytes' });
  }

  try {
    const hashed = await bcrypt.hash(password, SALT_ROUNDS);
    const user = await prisma.user.create({
      data: { username: normalizedUsername, password: hashed },
      select: { id: true, isAdmin: true },
    });
    return res.set('Cache-Control', 'no-store').status(201).json({ message: 'Registered', userId: user.id, token: issueToken(user) });
  } catch (error) {
    if (error.code === 'P2002') return res.status(400).json({ error: 'Username already taken' });
    next(error);
  }
};

exports.login = async (req, res, next) => {
  const { username, password } = req.body ?? {};
  if (typeof username !== 'string' || !username || username.length > 1024 || username.includes('\0') ||
      !isExistingPassword(password)) {
    return res.status(401).json({ error: 'Invalid credentials' });
  }

  try {
    const select = { id: true, password: true, isAdmin: true };
    const normalizedUsername = username.trim();
    let user = await prisma.user.findUnique({ where: { username }, select });
    if (!user && normalizedUsername && normalizedUsername !== username) {
      user = await prisma.user.findUnique({ where: { username: normalizedUsername }, select });
    }
    const matches = await bcrypt.compare(password, user?.password ?? DUMMY_HASH);
    if (!user?.password || !matches) return res.status(401).json({ error: 'Invalid credentials' });
    return res.set('Cache-Control', 'no-store').json({ message: 'Logged in', token: issueToken(user), admin: user.isAdmin });
  } catch (error) {
    next(error);
  }
};

exports.logout = (req, res) => {
  // Bearer tokens are stateless. The client clears its session on logout.
  res.set('Cache-Control', 'no-store').json({ message: 'Выход выполнен' });
};

exports.edit_pass = async (req, res, next) => {
  const { currentPassword, newPassword } = req.body ?? {};
  if (!isExistingPassword(currentPassword) || !isNewPassword(newPassword)) {
    return res.status(400).json({ error: 'Укажите текущий пароль и новый пароль от 8 символов до 72 байт' });
  }

  try {
    const { userId } = req.user;
    const user = await prisma.user.findUnique({ where: { id: userId }, select: { password: true } });
    if (!user) return res.status(404).json({ error: 'Пользователь не найден' });
    if (!user.password || !await bcrypt.compare(currentPassword, user.password)) {
      return res.status(401).json({ error: 'Текущий пароль неверен' });
    }

    const hashed = await bcrypt.hash(newPassword, SALT_ROUNDS);
    const { count } = await prisma.$transaction(async (tx) => {
      const result = await tx.user.updateMany({
        where: { id: userId, password: user.password },
        data: { password: hashed },
      });
      if (result.count) await tx.telegramLinkCode.deleteMany({ where: { userId } });
      return result;
    });
    if (!count) return res.status(409).json({ error: 'Пароль изменился. Повторите попытку.' });
    return res.json({ message: 'Пароль успешно изменен' });
  } catch (error) {
    next(error);
  }
};

const express = require('express');
const cors = require('cors');
const morgan = require('morgan');
const path = require('node:path');
const multer = require('multer');

function createApp({ logRequests = true } = {}) {
  const app = express();
  app.disable('x-powered-by');
  const proxyHops = Number(process.env.TRUST_PROXY_HOPS);
  if (Number.isInteger(proxyHops) && proxyHops > 0) app.set('trust proxy', proxyHops);

  const allowedOrigins = new Set([
    'https://muisicunderlab.vercel.app',
    'https://underlab-phi.vercel.app',
    'http://localhost:5173',
    'http://127.0.0.1:5173',
    'http://localhost:5180',
    'http://127.0.0.1:5180',
    ...(process.env.CORS_ORIGINS ?? '').split(',').map((origin) => origin.trim()).filter(Boolean),
  ]);

  app.use(cors({
    origin: (origin, callback) => callback(null, !origin || allowedOrigins.has(origin)),
    methods: ['GET', 'HEAD', 'POST', 'PUT', 'DELETE', 'OPTIONS'],
    allowedHeaders: ['Content-Type', 'Range', 'Authorization'],
    exposedHeaders: ['Content-Range', 'Accept-Ranges', 'Content-Length'],
  }));
  app.use(express.json({ limit: '64kb' }));
  app.use(express.urlencoded({ extended: false, limit: '64kb' }));
  if (logRequests) app.use(morgan(':method :status :response-time ms'));

  app.use('/images', express.static(path.join(__dirname, 'public', 'images')));
  app.use('/avatars', express.static(path.join(__dirname, 'public', 'avatars')));
  app.use('/auth', require('./routes/authRoutes'));
  app.use('/users', require('./routes/userRoutes'));
  app.use('/tracks', require('./routes/trackRoutes'));
  app.use('/playback', require('./routes/playbackRoutes'));
  app.use('/favorites', require('./routes/favoriteRoutes'));
  app.use('/complaints', require('./routes/complaintRoutes'));
  app.use('/reviews', require('./routes/reviewRoutes'));
  app.use('/api/admin', require('./routes/adminRoutes'));

  app.use((req, res) => res.status(404).json({ error: 'Not found' }));
  app.use((error, req, res, next) => {
    if (res.headersSent) return next(error);
    if (error instanceof multer.MulterError) {
      const tooLarge = error.code === 'LIMIT_FILE_SIZE' || error.code === 'LIMIT_FILE_COUNT';
      return res.status(tooLarge ? 413 : 400).json({
        error: tooLarge ? 'Upload exceeds the configured limit' : 'Invalid multipart upload',
      });
    }
    const status = error.status ?? error.statusCode;
    if (Number.isInteger(status) && status >= 400 && status < 500) {
      const message = status === 413 ? 'Request body is too large' : 'Invalid request';
      return res.status(status).json({ error: message });
    }
    console.error('Unhandled request error:', error.name);
    return res.status(500).json({ error: 'Internal server error' });
  });
  return app;
}

if (require.main === module) {
  if (!process.env.JWT_SECRET) throw new Error('JWT_SECRET is required');
  const port = process.env.PORT || 4000;
  createApp().listen(port, () => {
    console.log(`Сервер запущен на порту ${port}`);
    require('./bot/bot').launch();
  });
}

module.exports = { createApp };

model PlaybackDevice {
  userId     Int
  user       User     @relation(fields: [userId], references: [id], onDelete: Cascade)
  id         String   @db.Uuid
  keyHash    String   @db.Char(64)
  name       String
  platform   String
  lastSeenAt DateTime @default(now())

  @@id([userId, id])
  @@index([userId, lastSeenAt])
}

model Favorite {
  favoriteId Int      @id @default(autoincrement())
  trackId    Int
  userId     Int
  createdAt  DateTime @default(now())
  updatedAt  DateTime @updatedAt
  user       User     @relation(fields: [userId], references: [id])
  track      Track    @relation(fields: [trackId], references: [id])

  @@unique([userId, trackId])
  @@index([trackId])
}

model Complaint {
  id         Int      @id @default(autoincrement())
  userId     Int
  message    String
  status     String   @default("ожидание") // Возможные значения: pending, resolved, rejected
  createdAt  DateTime @default(now())
  updatedAt  DateTime @updatedAt
  user       User     @relation(fields: [userId], references: [id])

  @@index([userId])
}


model Review {
  id         Int      @id @default(autoincrement())
  userId     Int
  trackId    Int
  content    String
  rating     Int      // Рейтинг (1-5)
  createdAt  DateTime @default(now())
  updatedAt  DateTime @updatedAt
  user       User     @relation(fields: [userId], references: [id])
  track      Track    @relation(fields: [trackId], references: [id])

  @@index([trackId])
  @@index([userId])
}
